local function newTextPool()
    local pool, used = {}, 0
    local api = {}

    function api:begin()
        used = 0
    end

    function api:add(text, worldPosition, color, size, yOffset)
        used += 1

        local drawing = pool[used]
        if not drawing then
            drawing = Drawing.new("Text")
            drawing.Center = true
            drawing.Outline = true
            pool[used] = drawing
        end

        local position, onScreen = Camera:WorldToViewportPoint(worldPosition)
        drawing.Visible = onScreen

        if onScreen then
            drawing.Text = text
            drawing.Position = Vector2.new(position.X, position.Y - (yOffset or 20))
            drawing.Color = color
            drawing.Size = size or 14
        end
    end

    function api:finish()
        for index = used + 1, #pool do
            pool[index].Visible = false
        end
    end

    function api:clear()
        for _, drawing in ipairs(pool) do
            drawing:Remove()
        end
        table.clear(pool)
        used = 0
    end

    return api
end

        NoclipConnection = RunService.Stepped:Connect(function()

local function getClosestPlayerTarget()
    local root = getRoot()
    if not root then return nil end

    local closestPlayer, closestRoot, shortest = nil, nil, math.huge

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local targetRoot = player.Character:FindFirstChild("HumanoidRootPart")
            local targetHumanoid = player.Character:FindFirstChildOfClass("Humanoid")

            if targetRoot and targetHumanoid and targetHumanoid.Health > 0 then
                local distance = (root.Position - targetRoot.Position).Magnitude
                if distance < shortest then
                    shortest, closestPlayer, closestRoot = distance, player, targetRoot
                end
            end
        end
    end

    return closestPlayer, closestRoot, shortest
end

local function identifyKey(part)
    local haystack = string.lower(part.Name) .. " " .. (part.Parent and string.lower(part.Parent.Name) or "")

    if not string.find(haystack, "key") then return nil end
    if part.Transparency >= 1 then return nil end

    local colors = {
        white  = { name = "White Key",  color = Color3.fromRGB(255, 255, 255) },
        blue   = { name = "Blue Key",   color = Color3.fromRGB(0, 150, 255) },
        red    = { name = "Red Key",    color = Color3.fromRGB(255, 50, 50) },
        green  = { name = "Green Key",  color = Color3.fromRGB(0, 255, 100) },
        yellow = { name = "Yellow Key", color = Color3.fromRGB(255, 255, 0) },
        orange = { name = "Orange Key", color = Color3.fromRGB(255, 140, 0) },
        purple = { name = "Purple Key", color = Color3.fromRGB(180, 0, 255) },
        cyan   = { name = "Cyan Key",   color = Color3.fromRGB(0, 255, 255) },
    }

    for keyword, data in pairs(colors) do
        if string.find(haystack, keyword) then
            return data.name, data.color
        end
    end

    return "Key", Color3.fromRGB(255, 255, 255)
end

local MELEE_KEYWORDS = {
    "axe", "fire axe", "machete", "crowbar", "baseball bat", "pipe", "knife",

local function isMeleeWeapon(name)
    local lower = name:lower()
    for _, keyword in ipairs(MELEE_KEYWORDS) do
        if lower:find(keyword) then return true end
    end
    return false
end

local function TeleportTo(target)
    local root = getRoot()
    if not root then return end
    
    if typeof(target) == "CFrame" then
        root.CFrame = target
    elseif typeof(target) == "Vector3" then
        root.CFrame = CFrame.new(target)
    end
end
